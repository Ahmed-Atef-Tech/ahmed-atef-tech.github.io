# tidy.py: move finished exports into dated folders
import shutil, time
from pathlib import Path

inbox = Path("D:/exports")
while True:
    for f in inbox.glob("*.step"):
        day = inbox / time.strftime("%Y-%m-%d")
        day.mkdir(exist_ok=True)
        shutil.move(f, day / f.name)
    time.sleep(5)
